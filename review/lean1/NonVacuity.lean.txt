import BLF
import Solution
open BLF

-- bit 1 of x0 + x1 + x2 + x3 is BLF (direct witness)
example : IsBLF (fun x : Fin 4 → Bool => bitAt 1 (∑ i, bval (x i))) :=
  ⟨1, 0, fun _ => 1, fun x => by simp [lin]⟩

-- the same function given by an explicit truth table ("weight in {2,3}"), checked by the kernel
example : IsBLF (fun x : Fin 4 → Bool =>
    decide (2 ≤ (x 0).toNat + (x 1).toNat + (x 2).toNat + (x 3).toNat ∧
            (x 0).toNat + (x 1).toNat + (x 2).toNat + (x 3).toNat ≤ 3)) :=
  ⟨1, 0, fun _ => 1, by decide +kernel⟩

-- constant false is BLF
example : IsBLF (fun _ : Fin 4 → Bool => false) := ⟨0, 0, fun _ => 0, by decide +kernel⟩
-- constant true is BLF
example : IsBLF (fun _ : Fin 4 → Bool => true) := ⟨0, 1, fun _ => 0, by decide +kernel⟩

-- a negative K and a negative coefficient (two's complement): bit 0 of (-3 - x0) is not x0
example : bitAt 0 (-3) = true ∧ bitAt 1 (-3) = false ∧ bitAt 2 (-3) = true ∧ bitAt 5 (-1) = true := by
  decide +kernel

-- IsBLF is not trivially true: some 3-variable function is not BLF
example : ∃ f : (Fin 3 → Bool) → Bool, ¬ IsBLF f := by
  by_contra h
  push_neg at h
  have : Nat.card {f : (Fin 3 → Bool) → Bool // IsBLF f} = 256 := by
    rw [Nat.card_congr (Equiv.subtypeUnivEquiv h)]
    simp [Nat.card_eq_fintype_card, Fintype.card_fun, Fintype.card_bool]
  rw [blf_count_three] at this
  norm_num at this

-- the parallelogram hypothesis can hold with x, y, z, w pairwise distinct
example : ∃ x y z w : Fin 2 → Bool, x ≠ y ∧ x ≠ z ∧ x ≠ w ∧ y ≠ z ∧ y ≠ w ∧ z ≠ w ∧
    ∀ i, bval (x i) + bval (y i) = bval (z i) + bval (w i) :=
  ⟨![true, false], ![false, true], ![false, false], ![true, true], by decide +kernel⟩

-- the bound 1 is attained (the inequality is not trivially 0)
example : |lin 0 ![1, 1] ![true, false] / 2 ^ 1 + lin 0 ![1, 1] ![false, true] / 2 ^ 1
    - lin 0 ![1, 1] ![false, false] / 2 ^ 1 - lin 0 ![1, 1] ![true, true] / 2 ^ 1| = 1 := by
  decide +kernel

-- without the hypothesis the inequality fails, so the hypothesis is doing work
example : ¬ (|lin 0 ![1, 1] ![true, true] / 2 ^ 0 + lin 0 ![1, 1] ![true, true] / 2 ^ 0
    - lin 0 ![1, 1] ![false, false] / 2 ^ 0 - lin 0 ![1, 1] ![false, false] / 2 ^ 0| ≤ 1) := by
  decide +kernel

-- instantiating the headline theorems
example : |lin 7 ![3, -5, 2] ![true, false, true] / 2 ^ 2 + lin 7 ![3, -5, 2] ![false, true, false] / 2 ^ 2
    - lin 7 ![3, -5, 2] ![true, true, false] / 2 ^ 2 - lin 7 ![3, -5, 2] ![false, false, true] / 2 ^ 2| ≤ 1 :=
  floor_parallelogram 2 7 ![3, -5, 2] _ _ _ _ (by decide +kernel)

-- majority of 3 is BLF via threshold_isBLF
example : IsBLF (fun x : Fin 3 → Bool => decide (0 < -1 + ∑ i, (![1, 1, 1] : Fin 3 → ℤ) i * bval (x i))) :=
  threshold_isBLF (-1) ![1, 1, 1]

-- XOR with an affine function: bit 1 of (x0 + x1 + x2) XOR (1 + x0) is BLF
example : IsBLF (fun x : Fin 3 → Bool => xor (bitAt 1 (lin 0 (fun _ => 1) x)) (affB true ![true, false, false] x)) :=
  isBLF_xor_affB ⟨1, 0, fun _ => 1, fun _ => rfl⟩ true ![true, false, false]

-- affB is the parity: affB false (all true) is XOR of all inputs, for n = 3
example : ∀ x : Fin 3 → Bool, affB false (fun _ => true) x = xor (xor (x 0) (x 1)) (x 2) := by
  decide +kernel

-- count consistency: the n=1 count is all 4 functions
example : Nat.card ((Fin 1 → Bool) → Bool) = 4 := by
  simp [Nat.card_eq_fintype_card, Fintype.card_fun, Fintype.card_bool]
