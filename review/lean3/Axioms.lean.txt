import Lean
import BLF
import Solution

/-! Verifier's axiom collector: every constant whose name starts with `BLF` (theorems, defs and
auxiliary constants), checked with `Lean.collectAxioms`. -/

open Lean Elab Command

#eval show CommandElabM Unit from do
  let env ← getEnv
  let allowed : List Name := [``propext, ``Classical.choice, ``Quot.sound]
  let mut nThm := 0
  let mut nAll := 0
  let mut bad : Array (Name × Array Name) := #[]
  let mut used : NameSet := {}
  for (n, ci) in env.constants.toList do
    if (`BLF).isPrefixOf n then
      nAll := nAll + 1
      if ci matches .thmInfo _ then nThm := nThm + 1
      let axs ← Lean.collectAxioms n
      for a in axs do used := used.insert a
      if axs.any (fun a => !allowed.contains a) then
        bad := bad.push (n, axs)
  logInfo m!"BLF constants checked: {nAll} (theorems: {nThm}); axioms used overall: {used.toList}; offending: {bad.size} {bad.toList}"
